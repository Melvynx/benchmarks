    (() => {
      'use strict';

      // SI units are used throughout the solver. CSS pixels appear only in render().
      const FIXED = Object.freeze({gravity:9.81, brickWidth:0.50, brickHeight:0.24,
        brickFriction:0.72, brickRestitution:0.18, carMass:1250,
        carWidth:4.2, carHeight:1.45, groundFriction:0.85, pixelsPerMeter:80});
      const DEFAULTS = Object.freeze({speed:22, columns:12, rows:8, layers:2, brickMass:2.5});
      const settings = {...DEFAULTS};
      const canvas = document.getElementById('scene');
      const ctx = canvas.getContext('2d', {alpha:false});
      const stage = canvas.parentElement;
      const controls = ['speed','columns','rows','layers','brickMass'];
      const outputs = Object.fromEntries(controls.map(key => [key,document.getElementById(key+'Value')]));
      const telemetry = {speed:document.getElementById('speedReadout'), time:document.getElementById('timeReadout'), moving:document.getElementById('movingReadout')};
      const status = document.getElementById('status');
      const runPill = document.getElementById('runPill');
      const runLabel = document.getElementById('runLabel');
      let bodies=[],car,bricks=[],particles=[];
      let phase='ready',elapsed=0,accumulator=0,lastFrame=0,frameCount=0,settleTime=0;
      let impactAge=99,shake=0,skidStart=null,cameraX=0,hasImpacted=false;
      let viewWidth=1,viewHeight=1,scale=80,groundY=1;
      let visualSeed=1;
      const WALL_X=8.35;

      function body(kind,id,x,y,w,h,m,mu,e,layer=-1){
        return {kind,id,x,y,w,h,m,invM:1/m,inertia:m*(w*w+h*h)/12,
          invI:12/(m*(w*w+h*h)),mu,e,layer,a:0,vx:0,vy:0,omega:0,
          ux:1,uy:0,ox:0,oy:1,ex:w/2,ey:h/2,
          sleeping:kind==='brick',stillTime:0};
      }
      const ground={id:-1,kind:'ground',invM:0,invI:0,vx:0,vy:0,omega:0,mu:FIXED.groundFriction,e:0.02,x:0,y:0};

      function makeScene(launch=false){
        bodies=[];bricks=[];particles=[];elapsed=0;accumulator=0;settleTime=0;
        impactAge=99;shake=0;skidStart=null;cameraX=0;hasImpacted=false;visualSeed=1;
        car=body('car',0,3.13,FIXED.carHeight/2,FIXED.carWidth,FIXED.carHeight,
          FIXED.carMass,0,0.06);
        if(launch) car.vx=settings.speed;
        bodies.push(car);
        let id=1;
        for(let layer=0;layer<settings.layers;layer++){
          for(let row=0;row<settings.rows;row++){
            for(let column=0;column<settings.columns;column++){
              // Each layer represents a separate depth plane. Bricks in one plane
              // contact one another; the car and ground contact every plane.
              const offset=((row+layer)&1)*FIXED.brickWidth/2;
              const x=WALL_X+(column+0.5)*FIXED.brickWidth+offset+layer*0.012;
              const y=(row+0.5)*FIXED.brickHeight;
              const b=body('brick',id++,x,y,FIXED.brickWidth,FIXED.brickHeight,
                settings.brickMass,FIXED.brickFriction,FIXED.brickRestitution,layer);
              b.column=column;b.row=row;b.startX=x;b.startY=y;
              bricks.push(b);bodies.push(b);
            }
          }
        }
        phase=launch?'running':'ready';
        updateUI();
      }

      function updateUI(){
        const labels={speed:`${settings.speed} m/s`,columns:`${settings.columns} bricks`,
          rows:`${settings.rows} bricks`,layers:`${settings.layers} ${settings.layers===1?'layer':'layers'}`,
          brickMass:`${settings.brickMass.toFixed(1)} kg`};
        controls.forEach(key=>{
          const input=document.getElementById(key);
          input.value=settings[key];outputs[key].textContent=labels[key];
          input.style.setProperty('--fill',`${100*(settings[key]-Number(input.min))/(Number(input.max)-Number(input.min))}%`);
        });
        const count=settings.columns*settings.rows*settings.layers;
        document.getElementById('brickCount').textContent=`${count.toLocaleString()} bodies`;
        document.getElementById('wallMass').textContent=`${(count*settings.brickMass).toLocaleString(undefined,{maximumFractionDigits:1})} kg`;
        document.getElementById('composition').textContent=`${settings.columns} × ${settings.rows} × ${settings.layers}`;
        runPill.classList.toggle('active',phase==='running');
        runLabel.textContent=phase==='running'?'In progress':phase==='settled'?'Complete':'Ready';
        status.textContent=phase==='running'?(impactAge<0.6?'Impact registered':'Test in progress'):
          phase==='settled'?'Debris settled':'Awaiting test';
        status.classList.toggle('impact',phase==='running'&&impactAge<0.6);
        telemetry.speed.innerHTML=`${Math.hypot(car.vx,car.vy).toFixed(1)}<small>m/s</small>`;
        telemetry.time.innerHTML=`${elapsed.toFixed(2)}<small>s</small>`;
        const moving=phase==='ready'?0:bricks.reduce((n,b)=>n+(Math.hypot(b.vx,b.vy)>0.17||Math.abs(b.omega)>0.42),0);
        telemetry.moving.innerHTML=`${moving}<small>/ ${count}</small>`;
        return moving;
      }

      function orient(b){
        b.ux=Math.cos(b.a);b.uy=Math.sin(b.a);b.ox=-b.uy;b.oy=b.ux;
        b.ex=Math.abs(b.ux)*b.w/2+Math.abs(b.ox)*b.h/2;
        b.ey=Math.abs(b.uy)*b.w/2+Math.abs(b.oy)*b.h/2;
      }
      function radius(b,nx,ny){
        return b.w*0.5*Math.abs(nx*b.ux+ny*b.uy)+
          b.h*0.5*Math.abs(nx*b.ox+ny*b.oy);
      }

      function contact(a,b){
        const dx=b.x-a.x,dy=b.y-a.y;
        if(Math.abs(dx)>=a.ex+b.ex||Math.abs(dy)>=a.ey+b.ey)return null;
        const candidates=[[a.ux,a.uy],[a.ox,a.oy],[b.ux,b.uy],[b.ox,b.oy]];
        let penetration=Infinity,nx=0,ny=0;
        for(const axis of candidates){
          const overlap=radius(a,axis[0],axis[1])+radius(b,axis[0],axis[1])-
            Math.abs(dx*axis[0]+dy*axis[1]);
          if(overlap<=0) return null;
          if(overlap<penetration){penetration=overlap;const sign=dx*axis[0]+dy*axis[1]>=0?1:-1;
            nx=axis[0]*sign;ny=axis[1]*sign;}
        }
        const tx=-ny,ty=nx;
        const at=a.x*tx+a.y*ty,bt=b.x*tx+b.y*ty;
        const tangentMin=Math.max(at-radius(a,tx,ty),bt-radius(b,tx,ty));
        const tangentMax=Math.min(at+radius(a,tx,ty),bt+radius(b,tx,ty));
        const cn=((a.x*nx+a.y*ny+radius(a,nx,ny))+
          (b.x*nx+b.y*ny-radius(b,nx,ny)))*0.5;
        const ct=(tangentMin+tangentMax)*0.5;
        return {a,b,nx,ny,penetration,px:nx*cn+tx*ct,py:ny*cn+ty*ct};
      }

      function groundContact(b){
        if(b.y-b.ey>=0)return null;
        const ux=b.ux,uy=b.uy,vx=b.ox,vy=b.oy,hx=b.w/2,hy=b.h/2;
        const corners=[[-hx,-hy],[hx,-hy],[hx,hy],[-hx,hy]];
        let bottom=Infinity,contactX=b.x;
        for(const [lx,ly] of corners){
          const x=b.x+lx*ux+ly*vx,y=b.y+lx*uy+ly*vy;
          if(y<bottom-0.004){bottom=y;contactX=x;}
          else if(Math.abs(y-bottom)<0.004) contactX=(contactX+x)*0.5;
        }
        if(bottom>=0) return null;
        return {a:b,b:ground,nx:0,ny:-1,penetration:-bottom,px:contactX,py:0};
      }

      function broadphase(){
        const cells=new Map(),pairs=[],seen=new Set(),cellSize=0.72;
        for(const b of bodies){
          const left=Math.floor((b.x-b.ex-0.012)/cellSize),right=Math.floor((b.x+b.ex+0.012)/cellSize);
          const bottom=Math.floor((b.y-b.ey-0.012)/cellSize),top=Math.floor((b.y+b.ey+0.012)/cellSize);
          for(let x=left;x<=right;x++) for(let y=bottom;y<=top;y++){
            const key=`${x},${y}`;
            let list=cells.get(key);
            if(!list){list=[];cells.set(key,list)}
            for(const other of list){
              if(b.kind==='brick'&&other.kind==='brick'&&b.layer!==other.layer) continue;
              if(b.sleeping&&other.sleeping)continue;
              if(Math.abs(b.x-other.x)>b.ex+other.ex+0.05||
                Math.abs(b.y-other.y)>b.ey+other.ey+0.05)continue;
              const pairKey=other.id*4096+b.id;
              if(!seen.has(pairKey)){seen.add(pairKey);pairs.push([other,b]);}
            }
            list.push(b);
          }
        }
        return pairs;
      }

      function solve(c){
        const {a,b,nx,ny,px,py,penetration}=c;
        if(a.sleeping||b.sleeping){
          if(Math.hypot(a.vx-b.vx,a.vy-b.vy)<0.045&&penetration<0.004)return;
          a.sleeping=false;b.sleeping=false;a.stillTime=0;b.stillTime=0;
        }
        if(a.kind==='car'&&b.kind==='brick'&&!hasImpacted){
          hasImpacted=true;
          // Free-rolling tires have no longitudinal slip on approach. After
          // impact, locked tires use the specified tire/ground friction.
          a.mu=FIXED.groundFriction;
        }
        const invMass=a.invM+b.invM;
        if(invMass===0)return;
        // Split penetration by inverse mass; a slop prevents perpetual jitter.
        const correction=Math.min(0.28,Math.max(0,penetration-0.0007)*0.52/invMass);
        a.x-=nx*correction*a.invM;a.y-=ny*correction*a.invM;
        b.x+=nx*correction*b.invM;b.y+=ny*correction*b.invM;
        const rax=px-a.x,ray=py-a.y,rbx=px-b.x,rby=py-b.y;
        const avx=a.vx-a.omega*ray,avy=a.vy+a.omega*rax;
        const bvx=b.vx-b.omega*rby,bvy=b.vy+b.omega*rbx;
        const rvx=bvx-avx,rvy=bvy-avy;
        const normalSpeed=rvx*nx+rvy*ny;
        if(normalSpeed>0)return;
        const raCross=rax*ny-ray*nx,rbCross=rbx*ny-rby*nx;
        const effective=invMass+raCross*raCross*a.invI+rbCross*rbCross*b.invI;
        if(effective<1e-12)return;
        const bounce=normalSpeed<-0.65?Math.min(a.e,b.e):0;
        const j=-(1+bounce)*normalSpeed/effective;
        const ix=j*nx,iy=j*ny;
        a.vx-=ix*a.invM;a.vy-=iy*a.invM;a.omega-=raCross*j*a.invI;
        b.vx+=ix*b.invM;b.vy+=iy*b.invM;b.omega+=rbCross*j*b.invI;
        // Coulomb friction at the same contact, including angular velocity.
        const tvx=(b.vx-b.omega*rby)-(a.vx-a.omega*ray);
        const tvy=(b.vy+b.omega*rbx)-(a.vy+a.omega*rax);
        const tangentSpeed=tvx*(-ny)+tvy*nx;
        const tx=-ny,ty=nx;
        const raT=rax*ty-ray*tx,rbT=rbx*ty-rby*tx;
        const tangentMass=invMass+raT*raT*a.invI+rbT*rbT*b.invI;
        if(tangentMass<1e-12)return;
        const limit=Math.sqrt(a.mu*b.mu)*j;
        const jt=Math.max(-limit,Math.min(limit,-tangentSpeed/tangentMass));
        a.vx-=jt*tx*a.invM;a.vy-=jt*ty*a.invM;a.omega-=raT*jt*a.invI;
        b.vx+=jt*tx*b.invM;b.vy+=jt*ty*b.invM;b.omega+=rbT*jt*b.invI;
        if(a.kind==='car'&&b.kind==='brick'&&impactAge>0.4&&j>0.5){
          impactAge=0;shake=Math.min(8,2+Math.sqrt(j)*0.10);
          addDust(px,py,Math.min(22,6+Math.floor(Math.sqrt(j)*0.3)));
          skidStart=skidStart===null?car.x:skidStart;
        }
      }

      function physicsStep(dt){
        for(const b of bodies){
          if(b.sleeping)continue;
          b.vy-=FIXED.gravity*dt;
          b.x+=b.vx*dt;b.y+=b.vy*dt;b.a+=b.omega*dt;orient(b);
        }
        const pairs=broadphase();
        const iterations=bodies.length>900?3:4;
        const groundCandidates=bodies.filter(b=>!b.sleeping&&b.y-b.ey<0.055);
        for(let i=0;i<iterations;i++){
          // Ground first makes stacked contacts less likely to sink.
          for(const b of groundCandidates){const c=groundContact(b);if(c)solve(c)}
          for(const [a,b] of pairs){const c=contact(a,b);if(c)solve(c)}
        }
        for(const b of bricks){
          if(b.sleeping)continue;
          if(Math.hypot(b.vx,b.vy)<0.06&&Math.abs(b.omega)<0.15)b.stillTime+=dt;
          else b.stillTime=0;
          if(b.stillTime>0.65){b.sleeping=true;b.vx=0;b.vy=0;b.omega=0;}
        }
        if(impactAge<99)impactAge+=dt;
        shake*=0.94;
      }

      function advance(dt){
        let maxSpeed=Math.max(settings.speed,Math.abs(car.vx));
        for(const b of bricks)maxSpeed=Math.max(maxSpeed,Math.hypot(b.vx,b.vy));
        const substeps=Math.max(1,Math.min(4,Math.ceil(maxSpeed*dt/0.26)));
        for(let k=0;k<substeps;k++) physicsStep(dt/substeps);
        elapsed+=dt;
        const moving=updateUI();
        if(elapsed>2.2&&moving===0&&Math.hypot(car.vx,car.vy)<0.12) settleTime+=dt;
        else settleTime=0;
        if(settleTime>0.8){phase='settled';updateUI()}
      }

      function visualRandom(){visualSeed=(visualSeed*1664525+1013904223)>>>0;return visualSeed/4294967296}
      function addDust(x,y,count){
        for(let i=0;i<count;i++){
          const angle=visualRandom()*Math.PI*1.25+0.1;
          const speed=0.6+visualRandom()*3;
          particles.push({x,y,vx:Math.cos(angle)*speed,vy:Math.sin(angle)*speed,
            size:1.5+visualRandom()*3.6,age:0,life:0.42+visualRandom()*0.65});
        }
        if(particles.length>140)particles.splice(0,particles.length-140);
      }
      function updateParticles(dt){
        for(let i=particles.length-1;i>=0;i--){
          const p=particles[i];p.age+=dt;
          if(p.age>=p.life){particles.splice(i,1);continue}
          p.vy-=2.5*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;
        }
      }

      function resize(){
        const rect=stage.getBoundingClientRect(),dpr=Math.min(window.devicePixelRatio||1,2);
        viewWidth=Math.max(1,rect.width);viewHeight=Math.max(1,rect.height);
        canvas.width=Math.round(viewWidth*dpr);canvas.height=Math.round(viewHeight*dpr);
        canvas.style.width=viewWidth+'px';canvas.style.height=viewHeight+'px';
        ctx.setTransform(dpr,0,0,dpr,0,0);
        const wallRight=WALL_X+(settings.columns+0.5)*FIXED.brickWidth+(settings.layers-1)*0.012;
        scale=Math.min(FIXED.pixelsPerMeter,(viewWidth-38)/(wallRight+2.7),
          (viewHeight*0.63)/(settings.rows*FIXED.brickHeight+0.45));
        scale=Math.max(12,scale);
        groundY=Math.min(viewHeight*0.75,viewHeight-110);
      }

      function drawEffects(){
        if(skidStart!==null){
          ctx.strokeStyle='#0c151bad';ctx.lineWidth=2;

          ctx.fillStyle=`rgba(201,174,145,${alpha})`;
          ctx.beginPath();ctx.arc(sx(p.x),sy(p.y),p.size*(1+p.age),0,Math.PI*2);ctx.fill();

          const alpha=(0.16-impactAge)/0.16;
          const glow=ctx.createRadialGradient(sx(WALL_X),sy(0.65),3,sx(WALL_X),sy(0.65),100);
          glow.addColorStop(0,`rgba(255,197,139,${alpha*0.20})`);

          ctx.fillStyle=glow;ctx.fillRect(sx(WALL_X)-100,sy(0.65)-100,200,200);

      function render(){
        ctx.save();
        if(shake>0.05){
          const dx=Math.sin(frameCount*2.23)*shake,dy=Math.cos(frameCount*1.79)*shake*0.35;
          ctx.translate(dx,dy);
        }
        background();
        // The rear depth planes are visible at their edges and after scattering.
        for(let layer=settings.layers-1;layer>=0;layer--){
          for(const b of bricks)if(b.layer===layer)drawBrick(b);
        }
        drawEffects();drawCar();
        ctx.restore();
      }

      function frame(now){
        const delta=Math.min(0.05,(now-lastFrame)/1000||0);lastFrame=now;frameCount++;
        if(phase==='running'){
          accumulator=Math.min(0.11,accumulator+delta);
          const fixedDt=1/90;
          let steps=0;
          while(accumulator>=fixedDt&&steps<10){advance(fixedDt);accumulator-=fixedDt;steps++;
            if(phase!=='running')break;}
          updateParticles(delta);
        }
        const visibleWorld=viewWidth/scale;
        const desired=Math.max(0,car.x-visibleWorld*0.52);
        cameraX+=(desired-cameraX)*Math.min(1,delta*3.5);
        render();requestAnimationFrame(frame);
      }
